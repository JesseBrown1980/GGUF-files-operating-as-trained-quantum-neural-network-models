def schur_norm_s20(n, k, y):
    """norm((I-P) H P) at 2^20, outward, by the Schur test on the edges that leave the block: the pair term from shells k-1 and k, the linear term from shell k, in the even basis."""

    def up(under, mul):
        return ((isqrt(under << 40) + 1) * mul.numerator) // mul.denominator + 1

    col_max = 0
    row = {}
    for q in (k - 1, k):
        n0 = n - q
        for m in range(q // 2 + 1):
            p = q - m
            edges = []
            if q + 2 > k and n0 >= 2:
                edges.append(((q + 2, m + 1), up((p + 1) * (m + 1) * n0 * (n0 - 1), Fr(1, n))))
            if q + 1 > k and n0 >= 1:
                edges.append(((q + 1, m), up((p + 1) * n0 * (2 if p == m else 1), abs(y))))
                if p > m:
                    edges.append(((q + 1, m + 1), up((m + 1) * n0 * (2 if p == m + 1 else 1), abs(y))))
            for target, value in edges:
                row[target] = row.get(target, 0) + value
            col_max = max(col_max, sum(value for _, value in edges))
    return isqrt(col_max * max(row.values())) + 1
